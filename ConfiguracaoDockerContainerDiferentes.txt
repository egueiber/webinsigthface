# Configuração docker com backend
# 1. Criar rede externa no docker
docker network create healthtrack_net
# 2. carregar o docker do processador de imagem (ver o nome no server)
docker compose -f docker-compose-unimedprocessador.yml up -d
# 3. carregar o docker do backend
docker compose -f docker_compose_backend.yml up -d
